"""Entscheidet, WANN ein Abgleich faellig ist - ohne Qt und ohne Dateizugriff.

Die Qt-Verdrahtung (QFileSystemWatcher, QTimer) liegt in
app/services/watch_service.py. Hier steht nur die Regel, und die ist testbar,
ohne eine Ereignisschleife laufen zu lassen.

Zwei Ausloeser, mit klarer Rangfolge:

* Der periodische Abgleich ist der PRIMAERMECHANISMUS. SMB-Freigaben melden
  Aenderungen, die andere Rechner vornehmen, grundsaetzlich nicht - ein Watcher
  allein wuerde solche Aenderungen nie sehen.
* Watcher-Ereignisse sind ein BESCHLEUNIGER fuer lokale Aenderungen. Sie werden
  entprellt gesammelt: waehrend ein Generator zweihundert Dateien schreibt,
  sollen nicht zweihundert Scans starten, sondern einer, wenn Ruhe eingekehrt
  ist.

Der Tracker kennt keine Zeitquelle. Jede Methode bekommt die aktuelle Zeit
uebergeben, damit Tests nicht warten muessen.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum


class SyncReason(str, Enum):
    WATCHER = "watcher"
    PERIODIC = "periodic"
    MANUAL = "manual"


@dataclass(frozen=True, slots=True)
class SyncRequest:
    reason: SyncReason
    paths: tuple[str, ...] = ()

    @property
    def is_targeted(self) -> bool:
        """Ein Watcher-Ereignis nennt betroffene Pfade, der Takt nicht."""
        return bool(self.paths)


@dataclass
class ChangeTracker:
    """Sammelt Aenderungsmeldungen und beantwortet: ist ein Abgleich faellig?"""

    debounce_seconds: float = 1.5
    periodic_seconds: float = 900.0
    watcher_enabled: bool = True

    _pending: set[str] = field(default_factory=set, init=False)
    _last_change: float | None = field(default=None, init=False)
    _last_sync: float | None = field(default=None, init=False)
    _suspended: bool = field(default=False, init=False)

    # Meldungen ----------------------------------------------------------------

    def note_change(self, path: str, now: float) -> None:
        """Eine Aenderung wurde gemeldet. Setzt das Sammelfenster zurueck.

        Zuruecksetzen statt starten: solange weitere Aenderungen eintreffen,
        wird gewartet. Sonst startet mitten in einem laufenden Schreibvorgang
        ein Scan, der halbe Dateien sieht.
        """
        if not self.watcher_enabled:
            return
        self._pending.add(path)
        self._last_change = now

    def note_sync_started(self, now: float) -> None:
        self._pending.clear()
        self._last_change = None
        self._last_sync = now
        self._suspended = True

    def note_sync_finished(self, now: float) -> None:
        self._last_sync = now
        self._suspended = False

    def suspend(self) -> None:
        """Waehrend ein Scan laeuft, wird nichts Neues faellig."""
        self._suspended = True

    def resume(self, now: float) -> None:
        self._suspended = False
        self._last_sync = now

    def reset(self, now: float) -> None:
        self._pending.clear()
        self._last_change = None
        self._last_sync = now
        self._suspended = False

    # Abfrage ------------------------------------------------------------------

    @property
    def pending_paths(self) -> tuple[str, ...]:
        return tuple(sorted(self._pending))

    @property
    def has_pending(self) -> bool:
        return bool(self._pending)

    def due(self, now: float) -> SyncRequest | None:
        """Der faellige Abgleich, oder None.

        Der Takt hat Vorrang vor den gesammelten Ereignissen: er deckt sie
        ohnehin mit ab, und ein Ereignis darf den Takt nicht endlos verschieben.
        """
        if self._suspended:
            return None

        if self._periodic_due(now):
            return SyncRequest(SyncReason.PERIODIC)

        if (
            self.watcher_enabled
            and self._pending
            and self._last_change is not None
            and now - self._last_change >= self.debounce_seconds
        ):
            return SyncRequest(SyncReason.WATCHER, self.pending_paths)

        return None

    def _periodic_due(self, now: float) -> bool:
        if self.periodic_seconds <= 0:
            return False
        if self._last_sync is None:
            return True
        return now - self._last_sync >= self.periodic_seconds

    def seconds_until_periodic(self, now: float) -> float:
        """Fuer die Statusanzeige: wie lange bis zum naechsten Takt."""
        if self.periodic_seconds <= 0:
            return float("inf")
        if self._last_sync is None:
            return 0.0
        return max(0.0, self.periodic_seconds - (now - self._last_sync))

"""WatchService - haelt die Bibliothek ohne Zutun aktuell.

Drei Aufgaben:

1. **Periodischer Abgleich** als Primaermechanismus. SMB meldet Aenderungen
   anderer Rechner grundsaetzlich nicht; ohne Takt wuerde eine Bibliothek, die
   von einem zweiten Rechner befuellt wird, nie aktuell.
2. **Datei-Watcher** als Beschleuniger fuer lokale Aenderungen. Bewusst nur auf
   den Wurzeln und dem gerade sichtbaren Ordner - ein Watcher ueber
   zehntausende Verzeichnisse sprengt die Handle-Grenzen von Windows.
3. **NAS-Pruefung** im Takt, damit der Circuit Breaker ueberhaupt etwas weiss.

Die entscheidende Regel: **die Pruefung laeuft NIEMALS im GUI-Thread.** Ein
os.stat auf eine tote SMB-Freigabe blockiert je nach Konfiguration zwanzig
Sekunden und laenger. Genau dagegen existiert der Breaker; ihn aus dem
GUI-Thread zu pruefen wuerde die Blockade herbeifuehren, die er verhindern soll.
Die Pruefung ist deshalb ein Job im io-Pool.

Die Entscheidungslogik (Entprellung, Faelligkeit) liegt Qt-frei in
app/filesystem/change_tracker.py.
"""

PROBE_GROUP = "nas_probe"


class NasProbeJob(Job):
    """Prueft die Erreichbarkeit der Wurzeln - im io-Pool, nicht im GUI-Thread."""

    pool = "io"
    title = "NAS-Erreichbarkeit wird geprueft"

    def __init__(self, health: NasHealthMonitor, roots: list[str], reporter) -> None:
        super().__init__(reporter)
        self._health = health
        self._roots = roots

    def run_job(self) -> dict[str, bool]:
        result: dict[str, bool] = {}
        for root in self._roots:
            self.check_cancelled()
            result[root] = self._health.probe(root)
        return result


class WatchService(QObject):
    """sync_requested(reason), nas_state_changed(dict)"""

    sync_requested = pyqtSignal(str)
    nas_state_changed = pyqtSignal(object)

    def __init__(
        self,
        index_service,
        health: NasHealthMonitor,
        jobs: JobManager,
        settings,
        reporter: ErrorReporter,
    ) -> None:
        super().__init__()
        self._index = index_service
        self._health = health
        self._jobs = jobs
        self._settings = settings
        self._reporter = reporter

        self._tracker = ChangeTracker(
            debounce_seconds=settings.get_int("index.watcher_debounce_ms") / 1000.0,
            periodic_seconds=float(settings.get_int("index.background_scan_interval_s")),
            watcher_enabled=settings.get_bool("index.watcher_enabled"),
        )

        self._watcher = QFileSystemWatcher(self)
        self._watcher.directoryChanged.connect(self._on_directory_changed)
        self._watcher.fileChanged.connect(self._on_directory_changed)

        self._tick = QTimer(self)
        self._tick.setInterval(TICK_MS)
        self._tick.timeout.connect(self._on_tick)

        self._probe_timer = QTimer(self)
        self._probe_timer.setInterval(
            max(5, settings.get_int("system.nas_probe_interval_s")) * 1000
        )
        self._probe_timer.timeout.connect(self.probe_now)

        self._last_nas_state: dict[str, str] = {}
        self._probing = False
        self._extra_paths: list[str] = []

        self._index.scan_finished.connect(self._on_scan_finished)
        settings.subscribe("index.watcher_enabled", self._on_watcher_setting)
        settings.subscribe("index.watcher_debounce_ms", self._on_debounce_setting)
        settings.subscribe("index.background_scan_interval_s", self._on_interval_setting)
        settings.subscribe("system.nas_probe_interval_s", self._on_probe_setting)
        settings.subscribe("library.roots", lambda _k, _v: self.refresh_paths())

    # Lebenszyklus -------------------------------------------------------------

    def start(self) -> None:
        self.refresh_paths()
        self._tracker.reset(time.monotonic())
        self._tick.start()
        self._probe_timer.start()
        self.probe_now()

    def stop(self) -> None:
        self._tick.stop()
        self._probe_timer.stop()
        self._clear_watch_paths()

    @property
    def tracker(self) -> ChangeTracker:
        return self._tracker

    def watched_paths(self) -> list[str]:
        return list(self._watcher.directories())

    # Pfade --------------------------------------------------------------------

    def refresh_paths(self) -> None:
        """Ueberwacht die Wurzeln und den zuletzt angesehenen Ordner.

        Nicht den ganzen Baum: Windows begrenzt die Zahl gleichzeitig
        ueberwachter Verzeichnisse, und bei einer Bibliothek mit tausenden
        Ordnern waere die Grenze sofort erreicht. Was der Watcher nicht sieht,
        holt der periodische Abgleich nach.
        """
        self._clear_watch_paths()
        wanted = [str(root) for root in self._settings.get_list("library.roots")]
        wanted.extend(self._extra_paths)

        for path in dict.fromkeys(wanted):
            if not path:
                continue
            # addPath meldet selbst, wenn der Pfad nicht existiert; ein nicht
            # erreichbares NAS ist hier kein Fehlerfall, sondern Normalbetrieb.
            self._watcher.addPath(path)

    def watch_folder(self, path: str) -> None:
        """Nimmt den gerade angesehenen Ordner zusaetzlich in die Ueberwachung."""
        if not path:
            return
        self._extra_paths = [path]
        self.refresh_paths()

    def _clear_watch_paths(self) -> None:
        directories = self._watcher.directories()
        if directories:
            self._watcher.removePaths(directories)
        files = self._watcher.files()
        if files:
            self._watcher.removePaths(files)

    # Ereignisse ---------------------------------------------------------------

    def _on_directory_changed(self, path: str) -> None:
        self._tracker.note_change(path, time.monotonic())

    def _on_tick(self) -> None:
        if self._index.scan_running:
            self._tracker.suspend()
            return

        request = self._tracker.due(time.monotonic())
        if request is None:
            return
        self._trigger(request)

    def _trigger(self, request: SyncRequest) -> None:
        now = time.monotonic()
        self._tracker.note_sync_started(now)
        if self._index.start_scan() is None:
            # Abgewiesen, etwa weil keine Wurzel konfiguriert ist. Der Takt
            # laeuft trotzdem weiter, sonst bliebe der Tracker fuer immer
            # ausgesetzt.
            self._tracker.note_sync_finished(now)
            return
        self.sync_requested.emit(request.reason.value)

    def _on_scan_finished(self, _stats) -> None:
        self._tracker.note_sync_finished(time.monotonic())

    # NAS ----------------------------------------------------------------------

    def probe_now(self) -> None:
        """Prueft die Erreichbarkeit aller Wurzeln im Hintergrund."""
        if self._probing:
            return
        roots = [str(root) for root in self._settings.get_list("library.roots")]
        if not roots:
            return

        self._probing = True
        job = NasProbeJob(self._health, roots, self._reporter)
        job.signals.result.connect(self._on_probe_result)
        job.signals.finished.connect(lambda _id, _state: self._clear_probing())
        self._jobs.submit(job, group=PROBE_GROUP)

    def _clear_probing(self) -> None:
        self._probing = False

    def _on_probe_result(self, _job_id: int, _payload: object) -> None:
        state = self._health.snapshot()
        if state != self._last_nas_state:
            self._last_nas_state = state
            self.nas_state_changed.emit(state)

    # Einstellungen ------------------------------------------------------------

    def _on_watcher_setting(self, _key: str, value: object) -> None:
        self._tracker.watcher_enabled = bool(value)
        if not value:
            self._clear_watch_paths()
        else:
            self.refresh_paths()

    def _on_debounce_setting(self, _key: str, value: object) -> None:
        self._tracker.debounce_seconds = int(value) / 1000.0

    def _on_interval_setting(self, _key: str, value: object) -> None:
        self._tracker.periodic_seconds = float(value)

    def _on_probe_setting(self, _key: str, value: object) -> None:
        self._probe_timer.setInterval(max(5, int(value)) * 1000)

    # Zustand ------------------------------------------------------------------

    def status(self) -> dict[str, object]:
        now = time.monotonic()

            "watched": len(self._watcher.directories()),
            "pending": len(self._tracker.pending_paths),
            "next_periodic_s": round(self._tracker.seconds_until_periodic(now)),
            "nas": self._health.snapshot(),
        }

    def reload(self) -> None:
        self._tree.clear()
        records = self._index.scan_history(200)
        for record in records:
            item = QTreeWidgetItem(
                [
                    _format_time(record.scan_time),
                    "vollstaendig" if record.scan_type == "full" else "inkrementell",
                    format(record.duration, ".2f") + " s",
                    str(record.new_files),
                    str(record.changed_files),
                    str(record.deleted_files),
                    _status_label(record.status, record.errors),
                ]
            )
            if record.note:
                item.setToolTip(0, record.note)
            self._tree.addTopLevelItem(item)
        for column in range(1, 7):
            self._tree.resizeColumnToContents(column)

        self._summary.setText(self._describe(records))

    def _describe(self, records) -> str:
        status = self._watch.status() if self._watch is not None else {}
        lines = []

        if status:
            watcher = "aktiv" if status.get("watcher_enabled") else "aus"
            lines.append(
                "Datei-Watcher: " + watcher
                + " auf " + str(status.get("watched", 0)) + " Ordnern"
                + "   |   naechster turnusmaessiger Abgleich in "
                + str(status.get("next_periodic_s", 0)) + " s"
            )
            nas = status.get("nas") or {}
            if nas:
                unreachable = [root for root, value in nas.items() if value == "open"]
                lines.append(
                    "NAS: " + ("nicht erreichbar: " + ", ".join(unreachable))
                    if unreachable
                    else "NAS: erreichbar (" + str(len(nas)) + " Wurzel(n))"
                )

        last = self._index.last_successful_scan()
        if last is not None:
            lines.append(
                "Letzter erfolgreicher Abgleich: " + _format_time(last.scan_time)
                + "   |   " + format(last.duration, ".2f") + " s"
                + "   |   " + (last.note or "keine Aenderungen")
            )
        else:
            lines.append("Noch kein erfolgreicher Abgleich protokolliert")

        lines.append(str(len(records)) + " Eintraege im Protokoll")
        return "\n".join(lines)

    def _probe(self) -> None:
        if self._watch is not None:
            self._watch.probe_now()


def _format_time(value: float) -> str:
    if not value:
        return ""
    return datetime.datetime.fromtimestamp(value).strftime("%d.%m.%Y %H:%M:%S")


def _status_label(status: str, errors: int) -> str:
    if status == "ok":
        return "erfolgreich"
    if status == "cancelled":
        return "abgebrochen"
    return "Fehler" + ((" (" + str(errors) + ")") if errors else "")
